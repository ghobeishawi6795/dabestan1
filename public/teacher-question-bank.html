function parseQuestions(htmlContent) { const parser = new DOMParser(); const doc = parser.parseFromString(htmlContent, 'text/html'); parsedQuestions = []; selectedQuestions.clear(); // حذف تمام script و style ها doc.querySelectorAll('script, noscript, style').forEach(s => s.remove()); const body = doc.body; const html = body.innerHTML; // روش ۱: جستجوی div با کلاس question let standardQuestions = doc.querySelectorAll('.question'); if (standardQuestions.length > 1) { standardQuestions.forEach((elem, index) => { elem.querySelectorAll('script, noscript, style').forEach(s => s.remove()); const title = elem.querySelector('.question-title')?.textContent || `سوال ${index + 1}`; parsedQuestions.push({ id: index + 1, title: title.trim(), content: elem.innerHTML, html: elem.outerHTML }); selectedQuestions.add(index); }); displayQuestionsPreview(); return; } // روش ۲: تقسیم بر اساس "مرحله X" (برای فایل‌های تعاملی فارسی) // الگو: "مرحله ۱" یا "مرحله 1" با emoji بعدش const stagePattern = /(مرحله\s*\d+\s*[^\n]*)/gi; const stageMatches = [...html.matchAll(stagePattern)]; if (stageMatches.length > 1) { let lastIndex = 0; stageMatches.forEach((match, i) => { const startIndex = match.index; const title = match[1].trim(); if (i > 0) { // محتوای بین آخرین مرحله و این مرحله const content = html.substring(lastIndex, startIndex); const tempDiv = document.createElement('div'); tempDiv.innerHTML = content; parsedQuestions.push({ id: i, title: title, content: tempDiv.innerHTML, html: tempDiv.outerHTML }); selectedQuestions.add(i - 1); } lastIndex = startIndex; }); // آخرین مرحله const lastContent = html.substring(lastIndex); const lastTempDiv = document.createElement('div'); lastTempDiv.innerHTML = lastContent; parsedQuestions.push({ id: stageMatches.length, title: stageMatches[stageMatches.length - 1][1].trim(), content: lastTempDiv.innerHTML, html: lastTempDiv.outerHTML }); selectedQuestions.add(stageMatches.length - 1); displayQuestionsPreview(); return; } // روش ۳: تقسیم بر اساس emoji‌های خاص const emojiPattern = /(|🟢|🔀||🧱|📍||🔎|🔐||🏆)/g; const emojiSections = html.split(emojiPattern); if (emojiSections.length > 10) { for (let i = 1; i < emojiSections.length; i += 2) { if (emojiSections[i] && emojiSections[i + 1]) { const emoji = emojiSections[i]; const content = emojiSections[i + 1]; const tempDiv = document.createElement('div'); tempDiv.innerHTML = `${emoji}${content}`; const text = tempDiv.textContent || ''; const lines = text.split('\n').filter(line => line.trim()); const title = lines[0]?.trim() || `بخش ${Math.floor(i / 2) + 1}`; parsedQuestions.push({ id: Math.floor(i / 2) + 1, title: title.substring(0, 50), content: tempDiv.innerHTML, html: tempDiv.outerHTML }); selectedQuestions.add(Math.floor(i / 2)); } } if (parsedQuestions.length > 1) { displayQuestionsPreview(); return; } } // روش ۴: تقسیم بر اساس headingها (h1-h6) const headings = body.querySelectorAll('h1, h2, h3, h4, h5, h6'); if (headings.length > 1) { headings.forEach((heading, index) => { let content = ''; let sibling = heading.nextElementSibling; while (sibling && !sibling.matches('h1, h2, h3, h4, h5, h6')) { content += sibling.outerHTML; sibling = sibling.nextElementSibling; } const tempDiv = document.createElement('div'); tempDiv.innerHTML = heading.outerHTML + content; parsedQuestions.push({ id: index + 1, title: heading.textContent.trim(), content: tempDiv.innerHTML, html: tempDiv.outerHTML }); selectedQuestions.add(index); }); displayQuestionsPreview(); return; } // روش ۵: تقسیم بر اساس hr const hrSections = html.split(//gi); if (hrSections.length > 1) { hrSections.forEach((chunk, index) => { if (chunk.trim()) { const tempDiv = document.createElement('div'); tempDiv.innerHTML = chunk; const text = tempDiv.textContent || ''; const title = text.split('\n')[0]?.trim() || `بخش ${index + 1}`; parsedQuestions.push({ id: index + 1, title: title.substring(0, 50), content: chunk, html: chunk }); selectedQuestions.add(index); } }); displayQuestionsPreview(); return; } // روش : کل محتوا یک سوال parsedQuestions.push({ id: 1, title: 'سوال', content: html, html: html }); selectedQuestions.add(0); displayQuestionsPreview(); }
