function parseQuestions(htmlContent) { const parser = new DOMParser(); const doc = parser.parseFromString(htmlContent, 'text/html'); parsedQuestions = []; selectedQuestions.clear(); // حذف تمام script و style ها doc.querySelectorAll('script, noscript, style').forEach(s => s.remove()); // روش ۱: جستجوی div با کلاس question let standardQuestions = doc.querySelectorAll('.question'); if (standardQuestions.length > 1) { standardQuestions.forEach((elem, index) => { elem.querySelectorAll('script, noscript, style').forEach(s => s.remove()); const title = elem.querySelector('.question-title')?.textContent || `سوال ${index + 1}`; parsedQuestions.push({ id: index + 1, title: title.trim(), content: elem.innerHTML, html: elem.outerHTML }); selectedQuestions.add(index); }); displayQuestionsPreview(); return; } // روش : جستجوی مراحل (برای فایل‌های تعاملی) // پیدا کردن تمام بخش‌هایی که با "مرحله" یا "stage" شروع می‌شن const body = doc.body; const allElements = body.querySelectorAll('*'); const stageElements = []; allElements.forEach(elem => { const text = elem.textContent || ''; // تشخیص مراحل با الگوهای مختلف if (text.includes('مرحله') || text.match(/مرحله\s*\d+/) || text.match(/stage\s*\d+/i)) { // بررسی کنیم که این المان والد یک مرحله است const parent = elem.closest('div, section, article') || elem; if (!stageElements.includes(parent)) { stageElements.push(parent); } } }); if (stageElements.length > 1) { stageElements.forEach((elem, index) => { elem.querySelectorAll('script, noscript, style').forEach(s => s.remove()); const text = elem.textContent || ''; const match = text.match(/مرحله\s*(\d+)/) || text.match(/stage\s*(\d+)/i); const title = match ? `مرحله ${match[1]}` : `مرحله ${index + 1}`; parsedQuestions.push({ id: index + 1, title: title, content: elem.innerHTML, html: elem.outerHTML }); selectedQuestions.add(index); }); displayQuestionsPreview(); return; } // روش ۳: تقسیم بر اساس emoji‌های خاص (، 🔀، 🧠 و...) const html = body.innerHTML; const emojiPattern = /(🎯|🟢|🔀||🧱|📍||🔎|🔐||🏆)/g; const sections = html.split(emojiPattern); if (sections.length > 10) { // یعنی چندین بخش با emoji داریم for (let i = 1; i < sections.length; i += 2) { if (sections[i] && sections[i + 1]) { const emoji = sections[i]; const content = sections[i + 1]; const tempDiv = document.createElement('div'); tempDiv.innerHTML = `${emoji}${content}`; // استخراج عنوان از متن const text = tempDiv.textContent || ''; const lines = text.split('\n').filter(line => line.trim()); const title = lines[0]?.trim() || `بخش ${Math.floor(i / 2) + 1}`; parsedQuestions.push({ id: Math.floor(i / 2) + 1, title: title.substring(0, 50), content: tempDiv.innerHTML, html: tempDiv.outerHTML }); selectedQuestions.add(Math.floor(i / 2)); } } if (parsedQuestions.length > 1) { displayQuestionsPreview(); return; } } // روش ۴: تقسیم بر اساس h1-h6 const headings = body.querySelectorAll('h1, h2, h3, h4, h5, h6'); if (headings.length > 1) { headings.forEach((heading, index) => { let content = ''; let sibling = heading.nextElementSibling; while (sibling && !sibling.matches('h1, h2, h3, h4, h5, h6')) { content += sibling.outerHTML; sibling = sibling.nextElementSibling; } const tempDiv = document.createElement('div'); tempDiv.innerHTML = heading.outerHTML + content; parsedQuestions.push({ id: index + 1, title: heading.textContent.trim(), content: tempDiv.innerHTML, html: tempDiv.outerHTML }); selectedQuestions.add(index); }); displayQuestionsPreview(); return; } // روش ۵: تقسیم بر اساس hr const hrSections = html.split(//gi); if (hrSections.length > 1) { hrSections.forEach((chunk, index) => { if (chunk.trim()) { const tempDiv = document.createElement('div'); tempDiv.innerHTML = chunk; const text = tempDiv.textContent || ''; const title = text.split('\n')[0]?.trim() || `بخش ${index + 1}`; parsedQuestions.push({ id: index + 1, title: title.substring(0, 50), content: chunk, html: chunk }); selectedQuestions.add(index); } }); displayQuestionsPreview(); return; } // روش ۶: کل محتوا یک سوال parsedQuestions.push({ id: 1, title: 'سوال', content: html, html: html }); selectedQuestions.add(0); displayQuestionsPreview(); }
